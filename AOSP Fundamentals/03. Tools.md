## 1. adbutils — ADB Automation

**What it is:** A Python wrapper around ADB that provides an object-oriented interface for device communication — no more shelling out to `subprocess`.

**Install:**
```bash
pip install adbutils
```

---

### Basic Device Operations

```python
from adbutils import adb

# List connected devices
for device in adb.device_list():
    print(device.serial, device.prop.name)

# Get specific device
d = adb.device("emulator-5554")

# Shell command
output = d.shell("getprop ro.build.version.release")
print(output)  # "14"

# Push/pull files
d.push("/local/path/vendor.img", "/data/local/tmp/vendor.img")
d.pull("/data/local/tmp/logcat.txt", "./logcat.txt")
```

### Automated SELinux Denial Monitor

```python
from adbutils import adb
import time
import re

def monitor_selinux_denials(serial: str, duration: int = 30):
    """Monitor AVC denials in real-time for a given duration."""
    d = adb.device(serial)
    d.shell("logcat -c")   # clear log

    print(f"Monitoring SELinux denials for {duration}s...")
    time.sleep(duration)

    log = d.shell("logcat -d -s 'auditd:E'")
    denials = []

    pattern = re.compile(r"avc: denied \{ (\w+) \} .* scontext=(\S+) tcontext=(\S+) tclass=(\w+)")

    for line in log.splitlines():
        m = pattern.search(line)
        if m:
            denials.append({
                "action":   m.group(1),
                "scontext": m.group(2),
                "tcontext": m.group(3),
                "tclass":   m.group(4),
                "raw":      line.strip(),
            })

    print(f"\nFound {len(denials)} denials:\n")
    for d in denials:
        print(f"  [{d['tclass']}] {d['scontext']} → {d['action']} → {d['tcontext']}")
        print(f"  Suggested fix: allow {d['scontext'].split(':')[2]} {d['tcontext'].split(':')[2]}:{d['tclass']} {{{d['action']}}};\n")

monitor_selinux_denials("emulator-5554", duration=10)
```

### HAL Process Monitor

```python
from adbutils import adb
import time

def monitor_hal_processes(serial: str, interval: int = 2):
    """Watch HAL processes — detect crashes and restarts."""
    d = adb.device(serial)
    previous = {}

    while True:
        raw = d.shell("ps -A | grep android.hardware")
        current = {}

        for line in raw.splitlines():
            parts = line.split()
            if len(parts) >= 9:
                pid  = parts[1]
                name = parts[-1]
                current[name] = pid

        # Detect restarts
        for name, pid in current.items():
            if name in previous and previous[name] != pid:
                print(f"[RESTART] {name}  old_pid={previous[name]}  new_pid={pid}")
            elif name not in previous:
                print(f"[NEW]     {name}  pid={pid}")

        for name in previous:
            if name not in current:
                print(f"[DIED]    {name}  was_pid={previous[name]}")

        previous = current
        time.sleep(interval)

monitor_hal_processes("emulator-5554")
```

### Find HIDL Entry Points in a HAL Binary

```python
from elftools.elf.elffile import ELFFile
from elftools.elf.sections import SymbolTableSection

def find_hidl_symbols(so_path: str):
    """Find HIDL-related exported symbols in a HAL .so."""
    with open(so_path, "rb") as f:
        elf = ELFFile(f)
        symtab = elf.get_section_by_name(".dynsym")

        if not isinstance(symtab, SymbolTableSection):
            print("No dynamic symbol table found.")
            return

        hidl_symbols = []
        for sym in symtab.iter_symbols():
            if sym.name and any(kw in sym.name for kw in [
                "HIDL_FETCH", "registerAsService",
                "IVibrator", "ICamera", "IPower"
            ]):
                hidl_symbols.append({
                    "name":    sym.name,
                    "value":   hex(sym["st_value"]),
                    "size":    sym["st_size"],
                    "binding": sym["st_info"]["bind"],
                    "type":    sym["st_info"]["type"],
                })

        print(f"Found {len(hidl_symbols)} HIDL symbols:\n")
        for s in hidl_symbols:
            print(f"  {s['name']}")
            print(f"    addr={s['value']}  size={s['size']}  bind={s['binding']}  type={s['type']}\n")

find_hidl_symbols("./android.hardware.vibrator@1.0-impl.so")
```

### Check SONAME and Dependencies

```python
from elftools.elf.elffile import ELFFile
from elftools.elf.dynamic import DynamicSection

def check_dependencies(so_path: str):
    """List SONAME and all DT_NEEDED dependencies of a .so."""
    with open(so_path, "rb") as f:
        elf = ELFFile(f)

        for section in elf.iter_sections():
            if not isinstance(section, DynamicSection):
                continue

            soname = None
            needed = []

            for tag in section.iter_tags():
                if tag.entry.d_tag == "DT_SONAME":
                    soname = tag.soname
                elif tag.entry.d_tag == "DT_NEEDED":
                    needed.append(tag.needed)

            print(f"SONAME: {soname}")
            print(f"\nDependencies ({len(needed)}):")
            for lib in sorted(needed):
                print(f"  {lib}")

check_dependencies("./android.hardware.camera.provider@2.4-impl.so")
```

### Find Unresolved Symbols (Missing Dependencies)

```python
from elftools.elf.elffile import ELFFile
from elftools.elf.sections import SymbolTableSection

def find_undefined_symbols(so_path: str):
    """List symbols that this .so requires from other libraries."""
    with open(so_path, "rb") as f:
        elf = ELFFile(f)
        dynsym = elf.get_section_by_name(".dynsym")
        if not isinstance(dynsym, SymbolTableSection):
            return

        undefined = [
            sym.name
            for sym in dynsym.iter_symbols()
            if sym["st_shndx"] == "SHN_UNDEF" and sym.name
        ]

        print(f"Undefined symbols ({len(undefined)}):")
        for name in sorted(undefined):
            print(f"  {name}")

find_undefined_symbols("./libhidlbase.so")
```

### Check Security Hardening Flags

```python
import lief

def check_security(path: str):
    """Check security hardening properties of a HAL binary."""
    b = lief.parse(path)

    checks = {
        "PIE (Position Independent)": b.is_pie,
        "NX (Non-Executable Stack)":  b.has_nx,
        "RELRO":                      any(
            seg.type == lief.ELF.SEGMENT_TYPES.GNU_RELRO
            for seg in b.segments
        ),
        "Stack Canary":               any(
            "__stack_chk_fail" in sym.name
            for sym in b.imported_symbols
        ),
        "FORTIFY_SOURCE":             any(
            "_chk" in sym.name
            for sym in b.imported_symbols
        ),
    }

    print(f"Security audit: {path}\n")
    for check, result in checks.items():
        status = "✓" if result else "✗"
        print(f"  [{status}] {check}")

check_security("/vendor/bin/hw/android.hardware.camera.provider@2.4-service")
```

### Strip Debug Symbols from a HAL Binary

```python
import lief
import shutil

def strip_debug_symbols(input_path: str, output_path: str):
    """Remove debug sections from a HAL binary (like strip command)."""
    b = lief.parse(input_path)

    debug_sections = [
        s for s in b.sections
        if s.name.startswith(".debug_") or s.name in [".comment", ".note"]
    ]

    print(f"Removing {len(debug_sections)} debug sections:")
    for s in debug_sections:
        print(f"  {s.name}  size={s.size}")
        b.remove(s)

    b.write(output_path)
    print(f"\nStripped binary written to: {output_path}")

strip_debug_symbols("./vibrator-service-debug", "./vibrator-service-stripped")
```

```python
from androguard.misc import AnalyzeAPK

def find_hal_calls(apk_path: str):
    """Find AIDL HAL interface usage in a system APK."""
    apk, dex_list, analysis = AnalyzeAPK(apk_path)

    hal_patterns = [
        "android/hardware/",
        "IVibrator",
        "IPower",
        "ICamera",
        "ServiceManager",
    ]

    print(f"HAL-related references in {apk_path}:\n")

    for cls in analysis.get_classes():
        class_name = cls.name
        for method in cls.get_methods():
            for _, call, _ in method.get_xref_to():
                call_str = str(call.class_name) + str(call.name)
                if any(p in call_str for p in hal_patterns):
                    print(f"  {class_name}->{method.name}()")
                    print(f"    calls → {call.class_name}->{call.name}\n")
```

### Detect Dangerous Permissions in Vendor APKs

```python
from androguard.misc import AnalyzeAPK
from pathlib import Path

DANGEROUS_PERMS = {
    "android.permission.CAMERA",
    "android.permission.RECORD_AUDIO",
    "android.permission.ACCESS_FINE_LOCATION",
    "android.permission.READ_CONTACTS",
    "android.permission.READ_PHONE_STATE",
}

def audit_permissions(apk_dir: str):
    """Scan a directory of APKs for dangerous permissions."""
    results = []

    for apk_path in Path(apk_dir).glob("*.apk"):
        try:
            apk, _, _ = AnalyzeAPK(str(apk_path))
            perms = set(apk.get_permissions())
            dangerous = perms & DANGEROUS_PERMS
            if dangerous:
                results.append((apk.get_package(), dangerous))
        except Exception as e:
            print(f"  [skip] {apk_path.name}: {e}")

    print(f"\nAPKs with dangerous permissions:\n")
    for pkg, perms in results:
        print(f"  {pkg}")
        for p in perms:
            print(f"    - {p}")

audit_permissions("/system/priv-app/")
```

### Parse Android Logcat Output

```python
import re
from rich.console import Console
from rich.table import Table
from loguru import logger

console = Console()

LOGCAT_PATTERN = re.compile(
    r"(\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d+)\s+"
    r"(\d+)\s+(\d+)\s+([VDIWEF])\s+"
    r"(\S+)\s*:\s+(.*)"
)

LEVEL_COLORS = {
    "V": "dim",
    "D": "blue",
    "I": "green",
    "W": "yellow",
    "E": "red",
    "F": "bold red",
}

def parse_logcat(log_path: str, filter_tag: str = None):
    """Parse a logcat file and display it as a rich table."""
    table = Table(title=f"Logcat: {log_path}", show_lines=False)
    table.add_column("Time",    style="cyan",  no_wrap=True)
    table.add_column("PID",     style="dim",   width=6)
    table.add_column("Level",   width=3)
    table.add_column("Tag",     style="bold",  width=30)
    table.add_column("Message")

    with open(log_path) as f:
        for line in f:
            m = LOGCAT_PATTERN.match(line)
            if not m:
                continue
            time, pid, tid, level, tag, msg = m.groups()
            if filter_tag and filter_tag not in tag:
                continue

            color = LEVEL_COLORS.get(level, "white")
            table.add_row(
                time, pid,
                f"[{color}]{level}[/{color}]",
                f"[{color}]{tag}[/{color}]",
                f"[{color}]{msg}[/{color}]",
            )

    console.print(table)

parse_logcat("./logcat.txt", filter_tag="android.hardware.vibrator")
```

### Real-Time Log Tail with Highlighting

```python
import subprocess
from rich.console import Console
from rich.text import Text

console = Console()

def tail_logcat(tag_filter: str = "android.hardware"):
    """Stream logcat with real-time color highlighting."""
    proc = subprocess.Popen(
        ["adb", "logcat", "-s", tag_filter],
        stdout=subprocess.PIPE,
        text=True,
    )

    for line in proc.stdout:
        line = line.rstrip()
        text = Text(line)

        if " E " in line or "FATAL" in line:
            text.stylize("bold red")
        elif " W " in line:
            text.stylize("yellow")
        elif " I " in line:
            text.stylize("green")
        elif "avc: denied" in line:
            text.stylize("bold magenta")
        else:
            text.stylize("dim")

        console.print(text)

tail_logcat()
```

### Build Log Analyzer

```python
import re
from pathlib import Path
from rich.console import Console
from rich.table import Table
from loguru import logger

console = Console()

def analyze_build_log(log_path: str):
    """Parse AOSP build log and summarize errors and warnings."""
    errors   = []
    warnings = []

    error_re   = re.compile(r"error: (.+)")
    warning_re = re.compile(r"warning: (.+)")
    file_re    = re.compile(r"^(\S+\.(?:cpp|h|hal|aidl|bp)):(\d+):")

    with open(log_path) as f:
        for i, line in enumerate(f, 1):
            em = error_re.search(line)
            wm = warning_re.search(line)
            fm = file_re.match(line)

            location = fm.group(0) if fm else f"line {i}"

            if em:
                errors.append((location, em.group(1)))
            elif wm:
                warnings.append((location, wm.group(1)))

    table = Table(title="Build Log Summary")
    table.add_column("Type",     width=8)
    table.add_column("Location", style="cyan")
    table.add_column("Message")

    for loc, msg in errors[:20]:
        table.add_row("[red]ERROR[/red]", loc, msg)
    for loc, msg in warnings[:20]:
        table.add_row("[yellow]WARN[/yellow]",  loc, msg)

    console.print(table)
    logger.info(f"Total: {len(errors)} errors, {len(warnings)} warnings")

analyze_build_log("./build.log")
```

### AOSP HAL Inspector CLI

```python
#!/usr/bin/env python3
"""aosp-tool: CLI for AOSP HAL inspection and automation."""

import typer
from adbutils import adb
from rich.console import Console
from rich.table import Table

app     = typer.Typer(help="AOSP HAL inspection toolkit")
console = Console()


@app.command()
def list_hals(
    serial: str = typer.Option("", help="Device serial (empty = first device)"),
    hidl:   bool = typer.Option(True,  help="Show HIDL HALs"),
    aidl:   bool = typer.Option(True,  help="Show AIDL HALs"),
):
    """List all registered HAL services on a connected device."""
    d = adb.device(serial) if serial else adb.device_list()[0]

    if aidl:
        table = Table(title="AIDL HAL Services")
        table.add_column("Service", style="cyan")
        for line in d.shell("service list | grep android.hardware").splitlines():
            if line.strip():
                table.add_row(line.strip())
        console.print(table)

    if hidl:
        table = Table(title="HIDL HAL Services")
        table.add_column("Interface",      style="cyan")
        table.add_column("Server PID",     style="green")
        for line in d.shell("lshal 2>/dev/null").splitlines()[1:]:
            parts = line.split()
            if len(parts) >= 4:
                table.add_row(parts[0], parts[3])
        console.print(table)


@app.command()
def check_vintf(
    serial: str = typer.Option("", help="Device serial"),
):
    """Check VINTF compatibility on the connected device."""
    d = adb.device(serial) if serial else adb.device_list()[0]
    result = d.shell("vintf --check-compat 2>&1")
    color  = "green" if "COMPATIBLE" in result else "red"
    console.print(f"[{color}]{result.strip()}[/{color}]")


@app.command()
def pull_manifest(
    serial:  str = typer.Option("", help="Device serial"),
    output:  str = typer.Option("manifest.xml", help="Output file"),
):
    """Pull VINTF manifest from device."""
    d = adb.device(serial) if serial else adb.device_list()[0]
    d.pull("/vendor/etc/vintf/manifest.xml", output)
```

```python
import xml.etree.ElementTree as ET
from pydantic import BaseModel, field_validator
from typing import Literal

class HalInterface(BaseModel):
    name:     str
    instance: str

class HalEntry(BaseModel):
    name:       str
    format:     Literal["hidl", "aidl"]
    version:    str
    transport:  str | None = None
    interfaces: list[HalInterface]

    @field_validator("version")
    @classmethod
    def version_format(cls, v):
        # HIDL: "1.0", AIDL: "1", "2", etc.
        if not v.replace(".", "").isdigit():
            raise ValueError(f"Invalid version format: {v}")
        return v

class VintfManifest(BaseModel):
    hals: list[HalEntry]

def parse_manifest(xml_path: str) -> VintfManifest:
    """Parse and validate a VINTF manifest.xml."""
    tree = ET.parse(xml_path)
    root = tree.getroot()

    hals = []
    for hal in root.findall("hal"):
        ifaces = []
        for iface in hal.findall("interface"):
            ifaces.append(HalInterface(
                name     = iface.findtext("name", ""),
                instance = iface.findtext("instance", ""),
            ))

        hals.append(HalEntry(
            name       = hal.findtext("name", ""),
            format     = hal.get("format", "hidl"),
            version    = hal.findtext("version", ""),
            transport  = hal.findtext("transport"),
            interfaces = ifaces,
        ))

    manifest = VintfManifest(hals=hals)
    print(f"Parsed {len(manifest.hals)} HAL entries\n")
    for h in manifest.hals:
        print(f"  [{h.format}] {h.name} v{h.version}")
        for i in h.interfaces:
            print(f"    /{i.name}/{i.instance}")

    return manifest
```

```python
import networkx as nx
import re
from pathlib import Path

def parse_selinux_policy(te_dir: str) -> nx.DiGraph:
    """Parse .te files and build an allow-rule graph."""
    G = nx.DiGraph()

    allow_re = re.compile(
        r"allow\s+(\w+)\s+(\w+):(\w+)\s+\{([^}]+)\}"
    )

    for te_file in Path(te_dir).glob("*.te"):
        content = te_file.read_text()
        for m in allow_re.finditer(content):
            src, tgt, cls, perms = m.groups()
            for perm in perms.split():
                G.add_edge(src, tgt, tclass=cls, perm=perm.strip())

    return G
```

### Module Build Order (Topological Sort)

```python
import networkx as nx

def compute_build_order(module_deps: dict[str, list[str]]) -> list[str]:
    """
    Given module → [dependencies], compute build order.
    module_deps: {"A": ["B", "C"], "B": ["C"], "C": []}
    """
    G = nx.DiGraph()
    for module, deps in module_deps.items():
        G.add_node(module)
        for dep in deps:
            G.add_edge(dep, module)   # dep must be built before module

    if not nx.is_directed_acyclic_graph(G):
        cycles = list(nx.simple_cycles(G))
        raise ValueError(f"Circular dependency detected: {cycles}")

    order = list(nx.topological_sort(G))
    print("Build order:")
    for i, module in enumerate(order, 1):
        print(f"  {i:2}. {module}")
    return order
```

```python
from perfetto.trace_processor import TraceProcessor

def analyze_hal_binder_latency(trace_path: str, hal_name: str = "vibrator"):
    """Find binder transactions to a specific HAL and report latency."""
    tp = TraceProcessor(trace=trace_path)

    result = tp.query(f"""
        SELECT
            slice.name,
            slice.ts,
            slice.dur,
            thread.name AS thread_name,
            process.name AS process_name
        FROM slice
        JOIN thread_track ON slice.track_id = thread_track.id
        JOIN thread       ON thread_track.utid = thread.utid
        JOIN process      ON thread.upid = process.upid
        WHERE
            slice.name LIKE '%{hal_name}%'
            AND slice.dur > 0
        ORDER BY slice.dur DESC
    """)

    rows = list(result)
    if not rows:
        print(f"No slices found for '{hal_name}'")
        return

    durations_ms = [r.dur / 1_000_000 for r in rows]
    avg = sum(durations_ms) / len(durations_ms)
    p99 = sorted(durations_ms)[int(len(durations_ms) * 0.99)]

    print(f"HAL Binder Latency Report: {hal_name}\n")
    print(f"  Calls analyzed : {len(rows)}")
    print(f"  Average latency: {avg:.3f} ms")
    print(f"  P99 latency    : {p99:.3f} ms")
    print(f"  Max latency    : {max(durations_ms):.3f} ms\n")
```

### Trace HAL sysfs Writes with eBPF

```python
from bcc import BPF

# eBPF C program — traces write() syscalls to sysfs LED paths
BPF_PROGRAM = """
#include <uapi/linux/ptrace.h>
#include <linux/fs.h>

BPF_HASH(write_counts, u32, u64);     // PID → write count
BPF_PERF_OUTPUT(events);

struct event_t {
    u32 pid;
    char comm[16];
    char filename[128];
    ssize_t count;
};

TRACEPOINT_PROBE(syscalls, sys_enter_write) {
    struct event_t event = {};
    u32 pid = bpf_get_current_pid_tgid() >> 32;

    bpf_get_current_comm(&event.comm, sizeof(event.comm));

    // Only trace HAL processes
    if (event.comm[0] != 'a') return 0;   // starts with 'a' (android.hardware.*)

    event.pid = pid;
    event.count = args->count;

    events.perf_submit(args, &event, sizeof(event));

    u64 *val = write_counts.lookup(&pid);
    u64  zero = 0;
    write_counts.lookup_or_try_init(&pid, &zero);
    (*write_counts.lookup(&pid))++;

    return 0;
}
"""

def trace_hal_writes():
    b = BPF(text=BPF_PROGRAM)
    print("Tracing HAL write() syscalls... Ctrl+C to stop\n")

    def handle_event(cpu, data, size):
        event = b["events"].event(data)
        print(f"  PID={event.pid:<6}  comm={event.comm.decode():<40}  bytes={event.count}")

    b["events"].open_perf_buffer(handle_event)

    try:
        while True:
            b.perf_buffer_poll()
    except KeyboardInterrupt:
        print("\nWrite counts by PID:")
        for pid, count in b["write_counts"].items():
            print(f"  PID {pid.value}: {count.value} writes")

trace_hal_writes()
```

### Trace HAL Process System Calls

```python
from bcc import BPF

SYSCALL_TRACER = """
#include <uapi/linux/ptrace.h>

BPF_HASH(syscall_counts, u64, u64);

TRACEPOINT_PROBE(raw_syscalls, sys_enter) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;

    // Filter by target PID (set from Python side)
    u32 target = TARGET_PID;
    if (pid != target) return 0;

    u64 id = args->id;
    u64 *val = syscall_counts.lookup_or_try_init(&id, &(u64){0});
    if (val) (*val)++;

    return 0;
}
"""

import subprocess
import time

def trace_hal_syscalls(hal_name: str = "android.hardware.vibrator", duration: int = 5):
    """Trace which syscalls a HAL process makes."""
    # Get PID of HAL process
    result = subprocess.run(
        ["adb", "shell", f"pidof {hal_name}"],
        capture_output=True, text=True
    )
    pid = int(result.stdout.strip())
    print(f"Tracing PID {pid} ({hal_name}) for {duration}s...\n")

    prog = SYSCALL_TRACER.replace("TARGET_PID", str(pid))
    b = BPF(text=prog)

    time.sleep(duration)

    SYSCALL_NAMES = {
        0: "read",    1: "write",   2: "open",    3: "close",
        9: "mmap",   10: "mprotect", 56: "clone",
        202: "futex", 257: "openat",
    }

    print("Syscall counts:")
    for syscall_id, count in sorted(b["syscall_counts"].items(), key=lambda x: -x[1].value):
        name = SYSCALL_NAMES.get(syscall_id.value, f"syscall_{syscall_id.value}")
        print(f"  {name:<20}: {count.value}")

trace_hal_syscalls()
```

### Binary Diff Tool for HAL Updates

```python
from construct import *
from dataclasses import dataclass

@dataclass
class DiffResult:
    offset: int
    old_byte: int
    new_byte: int

def binary_diff(path_a: str, path_b: str) -> list[DiffResult]:
    """Compare two HAL binaries byte-by-byte and report differences."""
    with open(path_a, "rb") as fa, open(path_b, "rb") as fb:
        a = fa.read()
        b = fb.read()

    diffs = []
    min_len = min(len(a), len(b))

    for i in range(min_len):
        if a[i] != b[i]:
            diffs.append(DiffResult(i, a[i], b[i]))

    print(f"Binary diff: {path_a} vs {path_b}")
    print(f"  Size A   : {len(a):,} bytes")
    print(f"  Size B   : {len(b):,} bytes")
    print(f"  Differences: {len(diffs)}\n")

    for diff in diffs[:30]:   # show first 30
        print(f"  offset={hex(diff.offset):>10}  "
              f"old={hex(diff.old_byte):>4}  "
              f"new={hex(diff.new_byte):>4}")

    if len(b) > len(a):
        print(f"\n  {len(b) - len(a)} extra bytes in B (appended)")

    return diffs
```
